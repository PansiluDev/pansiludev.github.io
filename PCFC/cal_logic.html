Carbon Footprint Calculator Logic

Car Emissions

// Calculate fuel consumed and emissions

litersConsumed = carKm * carL100 / 100

carEmissions = litersConsumed * (car_gasoline or car_diesel)

Flight Emissions

flightEmissions = flightKm * factor_for_flight_class

Electricity Emissions

// Convert monthly consumption to annual kWh

electricity_kwh = monthlyWatts * 24 * 30 / 1000

electricityEmissions = electricity_kwh * electricity_factor

Total Footprint

totalEmissions = carEmissions + flightEmissions + electricityEmissions + ...
