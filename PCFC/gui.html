import tkinter as tk
from tkinter import ttk, messagebox, scrolledtext


class CarbonCalculator:
    """
    A class to calculate a user's carbon footprint across multiple lifestyle categories,
    with regional emission factors and comparisons to average footprints.
    """

    def __init__(self, location: str):
        """
        Initialize the calculator with a user's location (for regional emission factors).
        :param location: User's region (e.g., "US", "EU", "India"). Case-insensitive.
        """
        self.location = location.upper()  # Normalize to uppercase for consistency
        self.emission_factors = self._load_emission_factors()
        self.user_data = {
            "transportation": {
                "car": {"miles": 0, "mpg": 0, "fuel": "gasoline"},
                "flights": {"miles": 0, "class": "economy"}
            },
            "home_energy": {
                "electricity_kwh": 0,
                "natural_gas_therms": 0
            },
            "diet": {
                "meat": {},  # Format: {"beef": 100g/day, "chicken": 50g/day}
                "food_waste_kg": 0  # Annual waste (kg)
            },
            "shopping": {
                # Format: [{"garment": "cotton_shirt", "count": 12}, ...]
                "fast_fashion": [],
                "electronics_recycled_kg": 0  # Annual recycled weight (kg)
            }
        }

    def _load_emission_factors(self) -> dict:
        """
        Load emission factors for different categories (region-specific where applicable).
        Data sources: EPA, IPCC, and hypothetical averages for demonstration.
        """
        return {
            # Transportation (kg CO₂ per unit)
            "gasoline_kg_per_gallon": 8.887,  # EPA 2023: Gasoline emits ~8.887 kg CO₂/gal
            "diesel_kg_per_gallon": 10.18,   # EPA 2023: Diesel emits ~10.18 kg CO₂/gal
            # EPA: Economy class ~0.1 kg CO₂/passenger-mile
            "flight_economy_kg_per_mile": 0.1,
            "flight_business_kg_per_mile": 0.2,  # Business class ~0.2 kg CO₂/passenger-mile
            "flight_first_kg_per_mile": 0.3,    # First class ~0.3 kg CO₂/passenger-mile

            # Home Energy (kg CO₂ per kWh or therm)
            "electricity_kg_per_kwh": self._get_region_electricity_factor(),  # Region-specific
            "natural_gas_kg_per_therm": 5.3,   # IPCC: ~5.3 kg CO₂/therm

            # Diet (kg CO₂ per kg of food eaten/wasted)
            "meat_emission_factors_kg_per_kg": {
                "beef": 65,    # IPCC: Beef ~65 kg CO₂/kg eaten
                "lamb": 50,    # Lamb ~50 kg CO₂/kg eaten
                "pork": 7,     # Pork ~7 kg CO₂/kg eaten
                "chicken": 6,  # Chicken ~6 kg CO₂/kg eaten
                "fish": 11     # Fish ~11 kg CO₂/kg eaten (average)
            },
            "food_waste_kg_per_kg": 0.5,  # Hypothetical: 0.5 kg CO₂/kg wasted

            # Shopping (kg CO₂ per garment or kg of electronics)
            "fast_fashion_factors_kg_per_garment": {
                "cotton_shirt": 2.1,    # Example: Cotton shirt ~2.1 kg CO₂
                "polyester_shirt": 3.5,  # Polyester shirt ~3.5 kg CO₂
                "pants": 5.0,           # Pants ~5.0 kg CO₂
                "jacket": 12.0          # Jacket ~12.0 kg CO₂
            },
            # Recycling reduces emissions by 0.5 kg CO₂/kg
            "electronics_recycling_kg_per_kg": -0.5
        }

    def _get_region_electricity_factor(self) -> float:
        """
        Fetch region-specific electricity emission factors (kg CO₂/kWh).
        Sources: EPA (US), EEA (EU), and hypothetical for other regions.
        """
        region_data = {
            "US": 0.45,    # 2023 US grid avg (EPA)
            "EU": 0.25,    # 2023 EU grid avg (EEA)
            "INDIA": 0.75,  # 2023 India grid avg (hypothetical)
            "DEFAULT": 0.5  # Default for unknown regions
        }
        return region_data.get(self.location, region_data["DEFAULT"])

    def update_car_data(self, miles: float, mpg: float, fuel: str = "gasoline"):
        """
        Update car-related data for footprint calculation.
        :param miles: Annual miles driven (int/float)
        :param mpg: Miles per gallon (int/float)
        :param fuel: Fuel type ("gasoline" or "diesel")
        """
        self.user_data["transportation"]["car"] = {
            "miles": miles,
            "mpg": mpg,
            "fuel": fuel.lower()
        }

    def update_flight_data(self, miles: float, class_type: str = "economy"):
        """
        Update flight-related data.
        :param miles: Annual total flight miles (int/float)
        :param class_type: Flight class ("economy", "business", "first")
        """
        self.user_data["transportation"]["flights"] = {
            "miles": miles,
            "class": class_type.lower()
        }

    def update_electricity_data(self, kwh: float):
        """Update annual electricity consumption (kWh)."""
        self.user_data["home_energy"]["electricity_kwh"] = kwh

    def update_natural_gas_data(self, therms: float):
        """Update annual natural gas consumption (therms)."""
        self.user_data["home_energy"]["natural_gas_therms"] = therms

    def update_meat_consumption(self, daily_grams: dict):
        """
        Update daily meat consumption (grams per day).
        :param daily_grams: Dict like {"beef": 100, "chicken": 50}
        """
        # Convert grams to kg and normalize keys (lowercase)
        self.user_data["diet"]["meat"] = {
            meat_type.lower(): grams / 1000  # Store as kg/day
            for meat_type, grams in daily_grams.items()
        }

    def update_food_waste(self, kg_annual: float):
        """Update annual food waste (kg)."""
        self.user_data["diet"]["food_waste_kg"] = kg_annual

    def update_fast_fashion(self, garments: list):
        """
        Update fast fashion purchases.
        :param garments: List of dicts like [{"garment": "cotton_shirt", "count": 12}, ...]
        """
        # Normalize garment names to lowercase
        self.user_data["shopping"]["fast_fashion"] = [
            {"garment": g["garment"].lower(), "count": g["count"]}
            for g in garments
        ]

    def update_electronics_recycling(self, kg_annual: float):
        """Update annual electronics recycled (kg)."""
        self.user_data["shopping"]["electronics_recycled_kg"] = kg_annual

    def calculate_car_emissions(self) -> float:
        """Calculate CO₂ emissions from car use (kg/year)."""
        car_data = self.user_data["transportation"]["car"]
        if car_data["miles"] == 0 or car_data["mpg"] == 0:
            return 0.0

        fuel_used_gallons = car_data["miles"] / car_data["mpg"]
        fuel_factor = self.emission_factors[f"{car_data['fuel']}_kg_per_gallon"]
        return round(fuel_used_gallons * fuel_factor, 2)

    def calculate_flight_emissions(self) -> float:
        """Calculate CO₂ emissions from flights (kg/year)."""
        flight_data = self.user_data["transportation"]["flights"]
        if flight_data["miles"] == 0:
            return 0.0

        class_key = f"flight_{flight_data['class']}_kg_per_mile"
        # Fallback to economy if class is unrecognized
        factor = self.emission_factors.get(
            class_key, self.emission_factors["flight_economy_kg_per_mile"])
        return round(flight_data["miles"] * factor, 2)

    def calculate_electricity_emissions(self) -> float:
        """Calculate CO₂ emissions from home electricity (kg/year)."""
        kwh = self.user_data["home_energy"]["electricity_kwh"]
        if kwh == 0:
            return 0.0
        return round(kwh * self.emission_factors["electricity_kg_per_kwh"], 2)

    def calculate_natural_gas_emissions(self) -> float:
        """Calculate CO₂ emissions from natural gas (kg/year)."""
        therms = self.user_data["home_energy"]["natural_gas_therms"]
        if therms == 0:
            return 0.0
        return round(therms * self.emission_factors["natural_gas_kg_per_therm"], 2)

    def calculate_meat_emissions(self) -> float:
        """Calculate CO₂ emissions from meat consumption (kg/year)."""
        total = 0.0
        meat_factors = self.emission_factors["meat_emission_factors_kg_per_kg"]
        daily_meat_kg = self.user_data["diet"]["meat"]

        for meat_type, kg_per_day in daily_meat_kg.items():
            if meat_type not in meat_factors:
                continue  # Skip unknown meat types
            # Annual kg eaten * emission factor per kg
            annual_meat_kg = kg_per_day * 365  # Days/year
            total += annual_meat_kg * meat_factors[meat_type]

        return round(total, 2)

    def calculate_food_waste_emissions(self) -> float:
        """Calculate CO₂ emissions from food waste (kg/year)."""
        waste_kg = self.user_data["diet"]["food_waste_kg"]
        if waste_kg == 0:
            return 0.0
        return round(waste_kg * self.emission_factors["food_waste_kg_per_kg"], 2)

    def calculate_fast_fashion_emissions(self) -> float:
        """Calculate CO₂ emissions from fast fashion purchases (kg/year)."""
        total = 0.0
        garment_factors = self.emission_factors["fast_fashion_factors_kg_per_garment"]
        garments = self.user_data["shopping"]["fast_fashion"]

        for item in garments:
            garment = item["garment"]
            count = item["count"]
            if garment not in garment_factors:
                continue  # Skip unknown garments
            total += count * garment_factors[garment]

        return round(total, 2)

    def calculate_electronics_recycling_emissions(self) -> float:
        """Calculate CO₂ reduction from electronics recycling (kg/year)."""
        recycled_kg = self.user_data["shopping"]["electronics_recycled_kg"]
        if recycled_kg == 0:
            return 0.0
        return round(recycled_kg * self.emission_factors["electronics_recycling_kg_per_kg"], 2)

    def get_total_emissions_kg(self) -> float:
        """Calculate total CO₂ emissions (kg/year), including reductions from recycling."""
        return round(
            self.calculate_car_emissions() +
            self.calculate_flight_emissions() +
            self.calculate_electricity_emissions() +
            self.calculate_natural_gas_emissions() +
            self.calculate_meat_emissions() +
            self.calculate_food_waste_emissions() +
            self.calculate_fast_fashion_emissions() +
            self.calculate_electronics_recycling_emissions(),
            2
        )

    def get_total_footprint_tons(self) -> float:
        """Convert total emissions to metric tons (1 ton = 1,000 kg)."""
        return round(self.get_total_emissions_kg() / 1000, 2)

    def get_emission_breakdown(self) -> dict:
        """Return a breakdown of emissions by category (kg/year)."""
        return {
            "car": self.calculate_car_emissions(),
            "flights": self.calculate_flight_emissions(),
            "electricity": self.calculate_electricity_emissions(),
            "natural_gas": self.calculate_natural_gas_emissions(),
            "meat": self.calculate_meat_emissions(),
            "food_waste": self.calculate_food_waste_emissions(),
            "fast_fashion": self.calculate_fast_fashion_emissions(),
            "electronics_recycling": self.calculate_electronics_recycling_emissions(),
            "total_kg": self.get_total_emissions_kg(),
            "total_tons": self.get_total_footprint_tons()
        }

    def compare_to_region(self, region: str = None) -> dict:
        """
        Compare user’s footprint to regional averages.
        :param region: Optional region to compare (defaults to user's location).
        """
        region = region.upper() if region else self.location
        region_averages = {
            "US": 16.0,    # Annual avg footprint (tons/year) - EPA
            "EU": 12.0,    # EC - 2023
            "INDIA": 1.8,  # World Bank estimate
            "DEFAULT": 10.0
        }
        avg = region_averages.get(region, region_averages["DEFAULT"])
        user_total = self.get_total_footprint_tons()
        difference = user_total - avg

        return {
            "region": region,
            "region_avg_tons": round(avg, 2),
            "user_footprint_tons": user_total,
            "difference_tons": round(difference, 2),
            "message": (
                f"Your footprint ({user_total} tons) is "
                f"{round(abs(difference), 2)} tons "
                f"{'above' if difference > 0 else 'below'} the "
                f"{region} average ({avg} tons/year)."
            )
        }


class CarbonCalculatorApp:
    def __init__(self, master):
        self.master = master
        master.title("🌿 Carbon Footprint Calculator")
        master.geometry("800x600")
        master.configure(bg="#1e1e1e")

        # ---------- Main Canvas & Scrollbar ----------
        self.canvas = tk.Canvas(master, bg="#1e1e1e", highlightthickness=0)
        self.scrollbar = ttk.Scrollbar(
            master, orient="vertical", command=self.canvas.yview)
        self.scrollable_frame = tk.Frame(self.canvas, bg="#1e1e1e")

        self.scrollable_frame.bind(
            "",
            lambda e: self.canvas.configure(
                scrollregion=self.canvas.bbox("all")
            )
        )

        self.canvas.create_window(
            (0, 0), window=self.scrollable_frame, anchor="nw")
        self.canvas.configure(yscrollcommand=self.scrollbar.set)

        self.canvas.pack(side="left", fill="both", expand=True)
        self.scrollbar.pack(side="right", fill="y")

        # ---------- Styles ----------
        self.style = ttk.Style()
        self.style.theme_use('clam')
        self.style.configure("TLabel", font=("Arial", 11),
                             background="#1e1e1e", foreground="#e0e0e0")
        self.style.configure("Header.TLabel", font=(
            "Arial", 16, "bold"), background="#1e1e1e", foreground="#00ff99")
        self.style.configure("TButton", font=(
            "Arial", 12, "bold"), foreground="#ffffff", background="#4caf50")
        self.style.map("TButton", background=[('active', '#45a049')])

        # ---------- Widgets ----------
        self.create_widgets()

    def create_widgets(self):
        row = 0

        tk.Label(self.scrollable_frame, text="🌿 Carbon Footprint Calculator", font=("Arial", 20, "bold"),
                 bg="#1e1e1e", fg="#00ff99").grid(row=row, column=0, columnspan=2, pady=15)
        row += 1

        # ---------------- Transport Frame ----------------
        self.transport_frame = tk.LabelFrame(self.scrollable_frame, text="🚗 Transportation", padx=10, pady=10,
                                             bg="#2e2e2e", fg="#00ffff", font=("Arial", 12, "bold"))
        self.transport_frame.grid(
            row=row, column=0, columnspan=2, sticky="ew", padx=20, pady=5)
        row += 1

        tk.Label(self.transport_frame, text="Region (US/EU/India)",
                 bg="#2e2e2e", fg="#e0e0e0").grid(row=0, column=0, sticky="w", pady=3)
        self.location_entry = tk.Entry(
            self.transport_frame, bg="#3c3c3c", fg="#ffffff")
        self.location_entry.grid(row=0, column=1, pady=3)

        tk.Label(self.transport_frame, text="Annual car miles", bg="#2e2e2e",
                 fg="#e0e0e0").grid(row=1, column=0, sticky="w", pady=3)
        self.car_miles_entry = tk.Entry(
            self.transport_frame, bg="#3c3c3c", fg="#ffffff")
        self.car_miles_entry.grid(row=1, column=1)

        tk.Label(self.transport_frame, text="Car MPG", bg="#2e2e2e",
                 fg="#e0e0e0").grid(row=2, column=0, sticky="w", pady=3)
        self.car_mpg_entry = tk.Entry(
            self.transport_frame, bg="#3c3c3c", fg="#ffffff")
        self.car_mpg_entry.grid(row=2, column=1)

        tk.Label(self.transport_frame, text="Fuel type (gasoline/diesel)",
                 bg="#2e2e2e", fg="#e0e0e0").grid(row=3, column=0, sticky="w", pady=3)
        self.car_fuel_entry = tk.Entry(
            self.transport_frame, bg="#3c3c3c", fg="#ffffff")
        self.car_fuel_entry.grid(row=3, column=1)

        tk.Label(self.transport_frame, text="Annual flight miles",
                 bg="#2e2e2e", fg="#e0e0e0").grid(row=4, column=0, sticky="w", pady=3)
        self.flight_miles_entry = tk.Entry(
            self.transport_frame, bg="#3c3c3c", fg="#ffffff")
        self.flight_miles_entry.grid(row=4, column=1)

        tk.Label(self.transport_frame, text="Flight class (economy/business/first)",
                 bg="#2e2e2e", fg="#e0e0e0").grid(row=5, column=0, sticky="w", pady=3)
        self.flight_class_entry = tk.Entry(
            self.transport_frame, bg="#3c3c3c", fg="#ffffff")
        self.flight_class_entry.grid(row=5, column=1)

        # ---------------- Home Energy Frame ----------------
        self.energy_frame = tk.LabelFrame(self.scrollable_frame, text="💡 Home Energy", padx=10, pady=10,
                                          bg="#2e2e2e", fg="#ffcc00", font=("Arial", 12, "bold"))
        self.energy_frame.grid(
            row=row, column=0, columnspan=2, sticky="ew", padx=20, pady=5)
        row += 1

        tk.Label(self.energy_frame, text="Electricity kWh/year", bg="#2e2e2e",
                 fg="#e0e0e0").grid(row=0, column=0, sticky="w", pady=3)
        self.electricity_entry = tk.Entry(
            self.energy_frame, bg="#3c3c3c", fg="#ffffff")
        self.electricity_entry.grid(row=0, column=1)

        tk.Label(self.energy_frame, text="Natural Gas therms/year",
                 bg="#2e2e2e", fg="#e0e0e0").grid(row=1, column=0, sticky="w", pady=3)
        self.natural_gas_entry = tk.Entry(
            self.energy_frame, bg="#3c3c3c", fg="#ffffff")
        self.natural_gas_entry.grid(row=1, column=1)

        # ---------------- Diet Frame ----------------
        self.diet_frame = tk.LabelFrame(self.scrollable_frame, text="🥩 Diet", padx=10, pady=10,
                                        bg="#2e2e2e", fg="#ff6600", font=("Arial", 12, "bold"))
        self.diet_frame.grid(row=row, column=0, columnspan=2,
                             sticky="ew", padx=20, pady=5)
        row += 1

        meats = ["beef", "chicken", "lamb", "pork", "fish"]
        self.meat_entries = {}
        for i, meat in enumerate(meats):
            tk.Label(self.diet_frame, text=f"{meat.capitalize()} g/day",
                     bg="#2e2e2e", fg="#e0e0e0").grid(row=i, column=0, sticky="w", pady=3)
            self.meat_entries[meat] = tk.Entry(
                self.diet_frame, bg="#3c3c3c", fg="#ffffff")
            self.meat_entries[meat].grid(row=i, column=1)

        tk.Label(self.diet_frame, text="Food waste kg/year", bg="#2e2e2e",
                 fg="#e0e0e0").grid(row=len(meats), column=0, sticky="w", pady=3)
        self.food_waste_entry = tk.Entry(
            self.diet_frame, bg="#3c3c3c", fg="#ffffff")
        self.food_waste_entry.grid(row=len(meats), column=1)

        # ---------------- Shopping Frame ----------------
        self.shopping_frame = tk.LabelFrame(self.scrollable_frame, text="🛍️ Shopping", padx=10, pady=10,
                                            bg="#2e2e2e", fg="#ff3399", font=("Arial", 12, "bold"))
        self.shopping_frame.grid(
            row=row, column=0, columnspan=2, sticky="ew", padx=20, pady=5)
        row += 1

        tk.Label(self.shopping_frame, text="Fast fashion purchases (garment:count)",
                 bg="#2e2e2e", fg="#e0e0e0").grid(row=0, column=0, columnspan=2, sticky="w")
        self.fast_fashion_entries = []
        for i in range(5):  # allow up to 5 garment types
            g_entry = tk.Entry(self.shopping_frame, width=20,
                               bg="#3c3c3c", fg="#ffffff")
            c_entry = tk.Entry(self.shopping_frame, width=10,
                               bg="#3c3c3c", fg="#ffffff")
            g_entry.grid(row=i+1, column=0, padx=5, pady=2)
            c_entry.grid(row=i+1, column=1, padx=5, pady=2)
            self.fast_fashion_entries.append((g_entry, c_entry))

        tk.Label(self.shopping_frame, text="Electronics recycled kg/year",
                 bg="#2e2e2e", fg="#e0e0e0").grid(row=6, column=0, sticky="w", pady=3)
        self.electronics_entry = tk.Entry(
            self.shopping_frame, bg="#3c3c3c", fg="#ffffff")
        self.electronics_entry.grid(row=6, column=1)

        # ---------- Calculate Button ----------
        self.calc_button = tk.Button(self.scrollable_frame, text="Calculate Footprint", command=self.calculate,
                                     bg="#4caf50", fg="white", font=("Arial", 14, "bold"))
        self.calc_button.grid(row=row, column=0, columnspan=2, pady=20)
        row += 1

        # ---------- Results ----------
        self.result_frame = tk.LabelFrame(self.scrollable_frame, text="📊 Results", padx=10, pady=10,
                                          bg="#2e2e2e", fg="#00ff99", font=("Arial", 12, "bold"))
        self.result_frame.grid(
            row=row, column=0, columnspan=2, sticky="ew", padx=20, pady=10)

        self.result_text = scrolledtext.ScrolledText(self.result_frame, height=20, font=("Arial", 12),
                                                     bg="#1e1e1e", fg="#00ff99", insertbackground="white")
        self.result_text.pack(fill="both", expand=True)

    def calculate(self):
        location = self.location_entry.get()
        calc = CarbonCalculator(location)
        try:
            # Transportation
            calc.update_car_data(float(self.car_miles_entry.get()), float(
                self.car_mpg_entry.get()), self.car_fuel_entry.get())
            calc.update_flight_data(
                float(self.flight_miles_entry.get()), self.flight_class_entry.get())
            # Home energy
            calc.update_electricity_data(float(self.electricity_entry.get()))
            calc.update_natural_gas_data(float(self.natural_gas_entry.get()))
            # Diet
            meat_data = {k: float(v.get())
                         for k, v in self.meat_entries.items()}
            calc.update_meat_consumption(meat_data)
            calc.update_food_waste(float(self.food_waste_entry.get()))
            # Shopping
            garments = []
            for g, c in self.fast_fashion_entries:
                garment_name = g.get().strip()
                if garment_name:
                    count = int(c.get())
                    garments.append({"garment": garment_name, "count": count})
            calc.update_fast_fashion(garments)
            electronics = float(self.electronics_entry.get() or 0)
            calc.update_electronics_recycling(electronics)
        except ValueError:
            messagebox.showerror(
                "Error", "Please enter valid numbers for all fields.")
            return

        breakdown = calc.get_emission_breakdown()
        self.result_text.delete(1.0, tk.END)
        self.result_text.insert(
            tk.END, f"🌱 Total Carbon Footprint: {breakdown['total_kg']} kg/year ({breakdown['total_tons']} tons)\n\n")
        self.result_text.insert(tk.END, "📌 Breakdown by Category (kg/year):\n")
        for cat, val in breakdown.items():
            if cat not in ["total_kg", "total_tons"]:
                self.result_text.insert(
                    tk.END, f"   {cat.replace('_', ' ').capitalize()}: {val} kg\n")


if __name__ == "__main__":
    root = tk.Tk()
    app = CarbonCalculatorApp(root)
    root.mainloop()
